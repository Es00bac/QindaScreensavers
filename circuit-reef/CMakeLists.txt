cmake_minimum_required(VERSION 3.20)
project(CircuitReef VERSION 1.1.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
include(CTest)
include(GNUInstallDirs)
find_package(Threads REQUIRED)
find_package(PkgConfig QUIET)
option(REEF_SANITIZERS "Address and undefined-behavior sanitizers" OFF)
option(REEF_BUILD_QT "Build the optional Qt 6 Quick embedding library" OFF)
option(REEF_BUILD_QML_MODULE "Install the QindaQt.CircuitReef QML module" OFF)
if(PkgConfig_FOUND)
  pkg_check_modules(CAIRO QUIET IMPORTED_TARGET cairo)
  pkg_check_modules(SDL2 QUIET IMPORTED_TARGET sdl2>=2.0.18)
endif()
add_library(reef-core src/world.cpp src/metrics.cpp src/renderer.cpp src/geometry.cpp)
target_include_directories(reef-core PUBLIC include compat)
target_link_libraries(reef-core PUBLIC Threads::Threads)
set_target_properties(reef-core PROPERTIES POSITION_INDEPENDENT_CODE ON)
if(TARGET PkgConfig::CAIRO)
  target_link_libraries(reef-core PUBLIC PkgConfig::CAIRO)
  target_compile_definitions(reef-core PUBLIC REEF_CAIRO_SYSTEM_HEADERS=1)
else()
  find_library(CAIRO_RUNTIME NAMES cairo libcairo.so.2 REQUIRED)
  target_link_libraries(reef-core PUBLIC "${CAIRO_RUNTIME}")
  message(WARNING "Cairo development package absent: using included Linux ABI declarations against ${CAIRO_RUNTIME}")
endif()
add_executable(circuit-reef src/main.cpp)
target_link_libraries(circuit-reef PRIVATE reef-core)
if(TARGET PkgConfig::SDL2)
  target_link_libraries(circuit-reef PRIVATE PkgConfig::SDL2)
  target_compile_definitions(circuit-reef PRIVATE REEF_SDL_SYSTEM_HEADERS=1)
else()
  find_library(SDL2_RUNTIME NAMES SDL2 libSDL2-2.0.so.0 REQUIRED)
  target_link_libraries(circuit-reef PRIVATE "${SDL2_RUNTIME}")
  message(WARNING "SDL2 development package absent: using included Linux ABI declarations against ${SDL2_RUNTIME}")
endif()
if(CMAKE_CXX_COMPILER_ID MATCHES "GNU|Clang")
  target_compile_options(reef-core PRIVATE -Wall -Wextra -Wpedantic)
  target_compile_options(circuit-reef PRIVATE -Wall -Wextra -Wpedantic)
  if(REEF_SANITIZERS)
    target_compile_options(reef-core PUBLIC -fsanitize=address,undefined -fno-omit-frame-pointer)
    target_link_options(reef-core PUBLIC -fsanitize=address,undefined)
  endif()
endif()
if(BUILD_TESTING)
  add_executable(reef-tests tests/test_reef.cpp)
  target_link_libraries(reef-tests PRIVATE reef-core)
  add_test(NAME reef-core-tests COMMAND reef-tests)
  add_test(NAME reef-native-smoke COMMAND circuit-reef --private --windowed --size 640x360 --quit-after 0.25)
  set_tests_properties(reef-native-smoke PROPERTIES ENVIRONMENT "SDL_VIDEODRIVER=dummy;SDL_RENDER_DRIVER=software" TIMEOUT 20)
  add_test(NAME reef-headless-smoke COMMAND circuit-reef --snapshot ${CMAKE_CURRENT_BINARY_DIR}/test-frame.png --size 640x360 --seed 42 --demo)
  add_test(NAME reef-invalid-input COMMAND circuit-reef --fps nonsense)
  set_tests_properties(reef-invalid-input PROPERTIES WILL_FAIL TRUE)
endif()
if(REEF_BUILD_QT)
  add_subdirectory(qt)
endif()
install(TARGETS circuit-reef RUNTIME DESTINATION ${CMAKE_INSTALL_BINDIR})
install(FILES integration/studio.qinda.CircuitReef.desktop DESTINATION ${CMAKE_INSTALL_DATADIR}/applications)
install(FILES assets/icon.svg DESTINATION ${CMAKE_INSTALL_DATADIR}/icons/hicolor/scalable/apps RENAME studio.qinda.CircuitReef.svg OPTIONAL)
install(DIRECTORY docs/ DESTINATION ${CMAKE_INSTALL_DATADIR}/doc/circuit-reef)

install(FILES README.md LICENSE DESTINATION ${CMAKE_INSTALL_DATADIR}/doc/circuit-reef)

find_package(Qt6 6.4 REQUIRED COMPONENTS Core Gui)
find_package(LayerShellQt REQUIRED)
target_include_directories(circuit-reef PRIVATE ${CMAKE_CURRENT_SOURCE_DIR}/../common)
target_link_libraries(circuit-reef PRIVATE Qt6::Core Qt6::Gui LayerShellQt::Interface)
