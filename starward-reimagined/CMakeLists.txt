cmake_minimum_required(VERSION 3.20)
project(StarwardReimagined VERSION 2.1.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
include(CTest)
include(GNUInstallDirs)
find_package(Threads REQUIRED)
option(STARWARD_SANITIZERS "Address/undefined-behavior sanitizers" OFF)
find_package(PkgConfig QUIET)
if(PkgConfig_FOUND)
 pkg_check_modules(SDL2 QUIET IMPORTED_TARGET sdl2)
 pkg_check_modules(CAIRO QUIET IMPORTED_TARGET cairo)
endif()
if(NOT CMAKE_SYSTEM_NAME STREQUAL "Linux")
 message(FATAL_ERROR "This version targets Linux. Its OpenGL linkage uses the Linux GL ABI.")
endif()
find_library(GL_LIBRARY NAMES GL libGL.so.1 REQUIRED)
if(NOT TARGET PkgConfig::SDL2)
 find_library(SDL_LIBRARY NAMES SDL2 libSDL2-2.0.so.0 REQUIRED)
 message(STATUS "SDL2 runtime ABI build (install development files for normal distro builds)")
endif()
if(NOT TARGET PkgConfig::CAIRO)
 find_library(CAIRO_LIBRARY NAMES cairo libcairo.so.2 REQUIRED)
 message(STATUS "Cairo runtime ABI build")
endif()
foreach(SHADER mesh.vert mesh.frag full.vert shadow.frag sky.frag blur.frag post.frag)
 string(REPLACE "." "_" KEY "${SHADER}")
 string(TOUPPER "${KEY}" KEY)
 file(READ "${CMAKE_CURRENT_SOURCE_DIR}/shaders/${SHADER}" "${KEY}")
 set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS "${CMAKE_CURRENT_SOURCE_DIR}/shaders/${SHADER}")
endforeach()
configure_file(shaders/shader_strings.hpp.in generated/shader_strings.hpp @ONLY)
add_library(starward-scene src/meshes.cpp src/models.cpp src/director.cpp src/metrics.cpp src/export.cpp)
target_include_directories(starward-scene PUBLIC include)
target_link_libraries(starward-scene PUBLIC Threads::Threads)
add_executable(starward src/main.cpp src/renderer.cpp)
target_include_directories(starward PRIVATE compat "${CMAKE_CURRENT_BINARY_DIR}/generated")
target_link_libraries(starward PRIVATE starward-scene "${GL_LIBRARY}")
if(TARGET PkgConfig::SDL2)
 target_link_libraries(starward PRIVATE PkgConfig::SDL2)
 target_compile_definitions(starward PRIVATE STARWARD_SYSTEM_SDL=1)
else()
 target_link_libraries(starward PRIVATE "${SDL_LIBRARY}")
endif()
if(TARGET PkgConfig::CAIRO)
 target_link_libraries(starward PRIVATE PkgConfig::CAIRO)
 target_compile_definitions(starward PRIVATE STARWARD_SYSTEM_CAIRO=1)
else()
 target_link_libraries(starward PRIVATE "${CAIRO_LIBRARY}")
endif()
foreach(TARGET starward-scene starward)
 target_compile_options(${TARGET} PRIVATE -Wall -Wextra -Wpedantic)
 if(STARWARD_SANITIZERS)
  target_compile_options(${TARGET} PUBLIC -fsanitize=address,undefined -fno-omit-frame-pointer)
  target_link_options(${TARGET} PUBLIC -fsanitize=address,undefined)
 endif()
endforeach()
if(BUILD_TESTING)
 add_executable(starward-tests tests/tests.cpp)
 target_link_libraries(starward-tests PRIVATE starward-scene)
 add_test(NAME scene-motion-mesh-metrics COMMAND starward-tests)
 add_test(NAME help COMMAND starward --help)
 add_test(NAME reject-invalid-options COMMAND starward --fps nan)
 set_tests_properties(reject-invalid-options PROPERTIES WILL_FAIL TRUE)
endif()
install(TARGETS starward RUNTIME DESTINATION ${CMAKE_INSTALL_BINDIR})
install(FILES integration/studio.qinda.Starward.desktop DESTINATION ${CMAKE_INSTALL_DATADIR}/applications)
install(FILES README.md LICENSE DESTINATION ${CMAKE_INSTALL_DATADIR}/doc/starward-reimagined)
install(DIRECTORY docs/ DESTINATION ${CMAKE_INSTALL_DATADIR}/doc/starward-reimagined)

# Native Wayland output assignment must use layer-shell; SDL xdg-shell hints
# are ignored by some compositors. Keep SDL for previews and offline capture.
find_package(Qt6 6.4 REQUIRED COMPONENTS Core Gui)
find_package(LayerShellQt REQUIRED)
target_include_directories(starward PRIVATE ${CMAKE_CURRENT_SOURCE_DIR}/../common)
target_link_libraries(starward PRIVATE Qt6::Core Qt6::Gui LayerShellQt::Interface)
