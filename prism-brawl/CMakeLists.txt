cmake_minimum_required(VERSION 3.20)
project(PrismBrawl VERSION 1.2.1 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
include(CTest)
include(GNUInstallDirs)
find_package(Threads REQUIRED)
option(PRISM_SANITIZERS "Address/undefined-behavior sanitizers" OFF)
find_package(PkgConfig QUIET)
if(PkgConfig_FOUND)
 pkg_check_modules(SDL2 QUIET IMPORTED_TARGET sdl2>=2.0.18)
 pkg_check_modules(CAIRO QUIET IMPORTED_TARGET cairo)
endif()
if(NOT CMAKE_SYSTEM_NAME STREQUAL "Linux")
 message(FATAL_ERROR "This version targets Linux. Its OpenGL linkage uses the Linux GL ABI.")
endif()
find_library(GL_LIBRARY NAMES GL libGL.so.1 REQUIRED)
if(NOT TARGET PkgConfig::SDL2)
 find_library(SDL_LIBRARY NAMES SDL2 libSDL2-2.0.so.0 REQUIRED)
 message(STATUS "SDL2 runtime ABI build (install development files for normal distro builds)")
endif()
if(NOT TARGET PkgConfig::CAIRO)
 find_library(CAIRO_LIBRARY NAMES cairo libcairo.so.2 REQUIRED)
 message(STATUS "Cairo runtime ABI build")
endif()
foreach(SHADER mesh.vert mesh.frag full.vert shadow.frag sky.frag blur.frag post.frag)
 string(REPLACE "." "_" KEY "${SHADER}")
 string(TOUPPER "${KEY}" KEY)
 file(READ "${CMAKE_CURRENT_SOURCE_DIR}/shaders/${SHADER}" "${KEY}")
 set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS "${CMAKE_CURRENT_SOURCE_DIR}/shaders/${SHADER}")
endforeach()
configure_file(shaders/shader_strings.hpp.in generated/shader_strings.hpp @ONLY)
add_library(brawl-scene src/meshes.cpp src/models.cpp src/animation.cpp src/battle.cpp src/stages.cpp src/startup_stage.cpp src/director.cpp src/export.cpp)
target_include_directories(brawl-scene PUBLIC include)
target_link_libraries(brawl-scene PUBLIC Threads::Threads)
add_executable(prism-brawl src/main.cpp src/renderer.cpp)
target_include_directories(prism-brawl PRIVATE compat "${CMAKE_CURRENT_BINARY_DIR}/generated")
target_link_libraries(prism-brawl PRIVATE brawl-scene "${GL_LIBRARY}")
if(TARGET PkgConfig::SDL2)
 target_link_libraries(prism-brawl PRIVATE PkgConfig::SDL2)
 target_compile_definitions(prism-brawl PRIVATE BRAWL_SYSTEM_SDL=1)
else()
 target_link_libraries(prism-brawl PRIVATE "${SDL_LIBRARY}")
endif()
if(TARGET PkgConfig::CAIRO)
 target_link_libraries(prism-brawl PRIVATE PkgConfig::CAIRO)
 target_compile_definitions(prism-brawl PRIVATE BRAWL_SYSTEM_CAIRO=1)
else()
 target_link_libraries(prism-brawl PRIVATE "${CAIRO_LIBRARY}")
endif()
foreach(TARGET brawl-scene prism-brawl)
 target_compile_options(${TARGET} PRIVATE -Wall -Wextra -Wpedantic)
 if(PRISM_SANITIZERS)
  target_compile_options(${TARGET} PUBLIC -fsanitize=address,undefined -fno-omit-frame-pointer)
  target_link_options(${TARGET} PUBLIC -fsanitize=address,undefined)
 endif()
endforeach()
if(BUILD_TESTING)
 add_executable(brawl-tests tests/tests.cpp)
 target_link_libraries(brawl-tests PRIVATE brawl-scene)
 add_test(NAME combat-motion-mesh COMMAND brawl-tests)
 add_executable(brawl-stress tests/stress.cpp)
 target_link_libraries(brawl-stress PRIVATE brawl-scene)
 add_test(NAME help COMMAND prism-brawl --help)
 add_test(NAME arena-list COMMAND prism-brawl --list-stages)
 foreach(STAGE prism garden rooftop bliss compile aurora azure)
  add_test(NAME stage-${STAGE} COMMAND prism-brawl --stage ${STAGE} --seed 41 --battle-log 1)
 endforeach()
 add_test(NAME reject-invalid-stage COMMAND prism-brawl --stage missing)
 set_tests_properties(reject-invalid-stage PROPERTIES WILL_FAIL TRUE)
 add_test(NAME reject-invalid-options COMMAND prism-brawl --fps nan)
 set_tests_properties(reject-invalid-options PROPERTIES WILL_FAIL TRUE)
endif()
install(TARGETS prism-brawl RUNTIME DESTINATION ${CMAKE_INSTALL_BINDIR})
install(FILES integration/studio.qinda.PrismBrawl.desktop DESTINATION ${CMAKE_INSTALL_DATADIR}/applications)
install(FILES README.md LICENSE DESTINATION ${CMAKE_INSTALL_DATADIR}/doc/prism-brawl)
install(DIRECTORY docs/ DESTINATION ${CMAKE_INSTALL_DATADIR}/doc/prism-brawl)

install(FILES assets/studio.qinda.PrismBrawl.svg DESTINATION ${CMAKE_INSTALL_DATADIR}/icons/hicolor/scalable/apps)

# Native Wayland output assignment must use layer-shell; SDL xdg-shell hints
# are ignored by some compositors. Keep SDL for previews and offline capture.
find_package(Qt6 6.4 REQUIRED COMPONENTS Core Gui)
find_package(LayerShellQt REQUIRED)
target_include_directories(prism-brawl PRIVATE ${CMAKE_CURRENT_SOURCE_DIR}/../common)
target_link_libraries(prism-brawl PRIVATE Qt6::Core Qt6::Gui LayerShellQt::Interface)

target_include_directories(brawl-scene PUBLIC ${CMAKE_CURRENT_SOURCE_DIR}/../common)

if(BUILD_TESTING)
 add_executable(sfx-tests ../common/tests/sfx_tests.cpp)
 target_include_directories(sfx-tests PRIVATE ../common)
 target_link_libraries(sfx-tests PRIVATE PkgConfig::SDL2)
 add_test(NAME synthesized-sound COMMAND sfx-tests)
 set_tests_properties(synthesized-sound PROPERTIES ENVIRONMENT "SDL_AUDIODRIVER=dummy")
endif()
